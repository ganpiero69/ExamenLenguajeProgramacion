CREATE DATABASE DB_EXAMENT1;
USE DB_EXAMENT1;

CREATE TABLE GENERO
(
	INT_IDGENERO INT AUTO_INCREMENT PRIMARY KEY,
    VAR_NOMBRE VARCHAR(100) NOT NULL
);

CREATE TABLE PELICULA 
(
	INT_IDPELICULA INT AUTO_INCREMENT PRIMARY KEY,
    VAR_NOMBRE VARCHAR(100) NOT NULL,
    VAR_DIRECTOR VARCHAR(45) NOT NULL, 
    DME_FECHAESTRENO DATETIME NOT NULL,
    INT_IDGENERO INT,
	FOREIGN KEY (INT_IDGENERO) REFERENCES GENERO (INT_IDGENERO)
);
